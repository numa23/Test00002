def スクリプト名取得():
    arglist = load('argv')
    my_script_name = arglist[0]
    return replace(my_script_name, '.txt', '')

def スクリプトID取得():
    スクリプト名 = スクリプト名取得()
    分割リスト = split(スクリプト名, '_')
    count = len(分割リスト)
    結果 = ''
    for index in range(count):
        分割 = 分割リスト[index]
        if isnumeric(分割):
            break
        分割 += '_'
        結果 = 結果 + 分割
    return replace(スクリプト名, 結果, '')
